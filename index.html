        // --- IA LOGIC ---

        async function generateTrip() {

            const dest = document.getElementById('f-dest').value;

            const days = document.getElementById('f-days').value;

            const type = document.getElementById('f-type').value;

            const lang = document.getElementById('f-lang').value;


            if (!dest) return alert("Siusplau, introdueix un destí!");


            const loader = document.getElementById('loader');

            const status = document.getElementById('loader-status');

            loader.classList.remove('hidden');


            try {

                // 1. GENERAR ITINERARI (Gemini)

                status.innerText = "Preparant rutes amb IA...";

                const prompt = `Crea un itinerari de viatge detallat de ${days} dies a ${dest}. El tipus de viatge és ${type} i el pressupost és ${selectedBudget}.

                L'itinerari ha d'estar en idioma "${lang}".

                Respon NOMÉS amb un JSON amb aquest format:

                {

                  "title": "Títol del viatge",

                  "description": "Breu descripció",

                  "days": [{"num": 1, "topic": "Tema del dia", "activities": [{"time": "Matí", "desc": "Activitat 1", "lat": 0, "lng": 0}]}],

                  "packing": ["Item 1", "Item 2"],

                  "essentials": ["Lloc 1", "Lloc 2"]

                }`;


                const genResponse = await fetch(`https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash-preview-09-2025:generateContent?key=${apiKey}`, {

                    method: 'POST',

                    headers: {'Content-Type': 'application/json'},

                    body: JSON.stringify({

                        contents: [{ parts: [{ text: prompt }] }],

                        generationConfig: { responseMimeType: "application/json" }

                    })

                });


                const genData = await genResponse.json();

                const itinerary = JSON.parse(genData.candidates[0].content.parts[0].text);


                // 2. GENERAR IMATGE (Imagen)

                status.innerText = "Capturant els colors del destí...";

                let imageUrl = "https://images.unsplash.com/photo-1503220317375-aaad61436b1b?q=80&w=1000";

                try {

                    const imgResponse = await fetch(`https://generativelanguage.googleapis.com/v1beta/models/imagen-4.0-generate-001:predict?key=${apiKey}`, {

                        method: 'POST',

                        body: JSON.stringify({

                            instances: { prompt: `Cinematic professional travel photography of ${dest}, landmark visible, sunny day, high resolution` },

                            parameters: { sampleCount: 1 }

                        })

                    });

                    const imgData = await imgResponse.json();

                    if (imgData.predictions?.[0]?.bytesBase64Encoded) {

                        imageUrl = `data:image/png;base64,${imgData.predictions[0].bytesBase64Encoded}`;

                    }

                } catch (e) { console.error("Error Imagen", e); }


                const newTrip = {

                    id: Date.now(),

                    dest,

                    days,

                    type,

                    imageUrl,

                    ...itinerary

                };


                trips.unshift(newTrip);

                localStorage.setItem('tm_trips', JSON.stringify(trips));

                

                openTrip(newTrip.id);

            } catch (error) {

                console.error(error);

                alert("S'ha produït un error al generar el viatge. Revisa la clau d'API.");

            } finally {

                loader.classList.add('hidden');

            }

        }


        function renderTrips() {

            const list = document.getElementById('trips-list');

            const empty = document.getElementById('empty-state');

            list.innerHTML = '';


            if (trips.length === 0) {

                empty.classList.remove('hidden');

                return;

            }

            empty.classList.add('hidden');


            trips.forEach(trip => {

                const card = document.createElement('div');

                card.className = "glass p-4 rounded-3xl flex gap-4 cursor-pointer hover:bg-white transition-all shadow-sm";

                card.onclick = () => openTrip(trip.id);

                card.innerHTML = `

                    <img src="${trip.imageUrl}" class="w-20 h-20 rounded-2xl object-cover shadow-md">

                    <div class="flex-1 flex flex-col justify-center">

                        <h4 class="font-bold text-slate-800">${trip.dest}</h4>

                        <p class="text-xs text-indigo-600 font-black uppercase tracking-widest">${trip.days} Dies · ${trip.type}</p>

                    </div>

                    <button onclick="event.stopPropagation(); deleteTrip(${trip.id})" class="p-2 self-center text-slate-300 hover:text-red-500">

                        <i data-lucide="trash-2" class="w-5 h-5"></i>

                    </button>

                `;

                list.appendChild(card);

            });

            lucide.createIcons();

        }


        function openTrip(id) {

            currentTrip = trips.find(t => t.id === id);

            if (!currentTrip) return;


            document.getElementById('detail-img').src = currentTrip.imageUrl;

            document.getElementById('detail-title').innerText = currentTrip.dest;

            document.getElementById('detail-subtitle').innerText = `${currentTrip.days} DIES · ${currentTrip.type.toUpperCase()}`;


            // ITINERARI

            const itinBox = document.getElementById('tab-itin');

            itinBox.innerHTML = '';

            currentTrip.days.forEach(d => {

                const dayDiv = document.createElement('div');

                dayDiv.innerHTML = `

                    <div class="flex items-center gap-3 mb-4">

                        <span class="bg-indigo-600 text-white w-10 h-10 rounded-xl flex items-center justify-center font-black">D${d.num}</span>

                        <h4 class="font-extrabold text-slate-800">${d.topic}</h4>

                    </div>

                    <div class="space-y-4 ml-5 border-l-2 border-indigo-100 pl-6 pb-6">

                        ${d.activities.map(a => `

                            <div class="bg-white p-4 rounded-2xl shadow-sm border border-slate-100">

                                <p class="text-[10px] font-black text-indigo-500 uppercase mb-1">${a.time}</p>

                                <p class="text-sm font-bold text-slate-700">${a.desc}</p>

                            </div>

                        `).join('')}

                    </div>

                `;

                itinBox.appendChild(dayDiv);

            });


            // PACKING

            const packBox = document.getElementById('packing-list');

            packBox.innerHTML = currentTrip.packing.map(item => `

                <div class="flex items-center gap-3">

                    <input type="checkbox" class="w-5 h-5 rounded-lg border-2 border-slate-200 text-indigo-600">

                    <span class="text-sm font-bold text-slate-600">${item}</span>

                </div>

            `).join('');


            // ESSENTIALS

            const essBox = document.getElementById('essentials-list');

            essBox.innerHTML = currentTrip.essentials.map(e => `<div class="flex items-center gap-2"><i data-lucide="map-pin" class="w-3 h-3"></i> ${e}</div>`).join('');


            lucide.createIcons();

            setTab('itin');

            changeView('detail');

            updateMateyBubble(`Benvingut a ${currentTrip.dest}! T'he preparat un itinerari genial.`);

        }

            localStorage.setItem('tm_trips', JSON.stringify(trips));

            renderTrips();

        }

        async function sendChatMessage() {

            const input = document.getElementById('chat-input');

            const msg = input.value.trim();

            if (!msg) return;


            const box = document.getElementById('chat-messages');

            box.innerHTML += `<div class="bg-white p-3 rounded-2xl rounded-tr-none text-sm font-bold text-slate-800 ml-8 text-right shadow-sm border border-slate-100">${msg}</div>`;

            input.value = '';

            box.scrollTop = box.scrollHeight;


            try {

                const response = await fetch(`https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash-preview-09-2025:generateContent?key=${apiKey}`, {

                    method: 'POST',

                    headers: {'Content-Type': 'application/json'},

                    body: JSON.stringify({

                        contents: [{ parts: [{ text: msg }] }],

                        systemInstruction: { parts: [{ text: `Ets Matey, una maleta assistent de viatges. Respon en català de forma breu i amigable. Tens context del viatge actual a ${currentTrip?.dest || 'un lloc desconegut'}.` }] }

                    })

                });

                const data = await response.json();

                const aiMsg = data.candidates[0].content.parts[0].text;

                box.innerHTML += `<div class="bg-indigo-50 p-3 rounded-2xl rounded-tl-none text-sm font-medium text-indigo-900 mr-8 shadow-sm">${aiMsg}</div>`;

            } catch (e) {

                box.innerHTML += `<div class="text-red-500 text-[10px] font-bold text-center">Error de connexió</div>`;

            }

            box.scrollTop = box.scrollHeight;

        }

        function initMap() {

            if (!currentTrip) return;

            

            // Si el mapa ja existia, el destruïm per recrear-lo

            if (map) { map.remove(); }


            // Busquem la primera coordenada vàlida

            let center = [41.3851, 2.1734]; // Default Barcelona

            currentTrip.days.forEach(d => {

                d.activities.forEach(a => {

                    if (a.lat && a.lng) center = [a.lat, a.lng];

                });

            });


            map = L.map('map').setView(center, 13);

            L.tileLayer('https://{s}.tile.openstreetmap.org/{z}/{x}/{y}.png').addTo(map);


            currentTrip.days.forEach(d => {

                d.activities.forEach(a => {

                    if (a.lat && a.lng) {

                        L.marker([a.lat, a.lng]).addTo(map)

                            .bindPopup(`<b>Dia ${d.num}</b><br>${a.desc}`);

                    }

                });

            });

        }
